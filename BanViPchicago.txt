local function makeDraggable(topbarObject, object)
    local dragging, dragInput, dragStart, startPos
    topbarObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = object.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    topbarObject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            object.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function addToggle(page, textVI, textEN, y, configKey, callback)
    local frame = Instance.new("TextButton", page)
    frame.Size = UDim2.new(1, -10, 0, 30)
    frame.Position = UDim2.fromOffset(5, y)
    frame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    frame.BackgroundTransparency = 0.2
    frame.Text = ""
    frame.BorderSizePixel = 0
    
    local fCorner = Instance.new("UICorner", frame)
    fCorner.CornerRadius = UDim.new(0, 6)
    local stroke = Instance.new("UIStroke", frame)
    stroke.Thickness = 1
    stroke.Color = Color3.fromRGB(200, 200, 200)

    local lbl = Instance.new("TextLabel", frame)
    lbl.Size = UDim2.new(1, -45, 1, 0)
    lbl.Position = UDim2.fromOffset(6, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = (currentLang == "VI") and textVI or textEN
    lbl.TextColor3 = Color3.fromRGB(255, 255, 255)
    lbl.TextSize = 10
    lbl.Font = Enum.Font.Code
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local switchBg = Instance.new("Frame", frame)
    switchBg.Size = UDim2.fromOffset(36, 18)
    switchBg.Position = UDim2.new(1, -40, 0.5, -9)
    switchBg.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    switchBg.BorderSizePixel = 0
    Instance.new("UICorner", switchBg).CornerRadius = UDim.new(1, 0)

    local switchDot = Instance.new("Frame", switchBg)
    switchDot.Size = UDim2.new(0, 14, 0, 14)
    switchDot.Position = UDim2.new(0, 2, 0.5, -7)
    switchDot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    switchDot.BorderSizePixel = 0
    Instance.new("UICorner", switchDot).CornerRadius = UDim.new(1, 0)

    local active = (configKey and configData[configKey]) or false
    
    local function updateVisual()
        if active then
            switchBg.BackgroundColor3 = Color3.fromRGB(240, 240, 240)
            switchDot.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
            switchDot.Position = UDim2.new(1, -16, 0.5, -7)
            stroke.Color = Color3.fromRGB(255, 255, 255)
        else
            switchBg.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
            switchDot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            switchDot.Position = UDim2.new(0, 2, 0.5, -7)
            stroke.Color = Color3.fromRGB(200, 200, 200)
        end
    end

    updateVisual()

    local function triggerToggle()
        active = not active
        updateVisual()
        if configKey then
            configData[configKey] = active
            saveConfig()
        end
        if callback then callback(active, frame) end
    end

    frame.MouseButton1Click:Connect(triggerToggle)

    table.insert(toggleRegistry, {
        Label = lbl,
        TextVI = textVI,
        TextEN = textEN
    })

    if active and callback then
        task.spawn(function()
            task.wait(0.2)
            callback(true, frame)
        end)
    end

    return frame
end

local function refreshCarTpList()
    for _, child in pairs(playerListContainer:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end
    
    local yOffset = 0
    local activePlayers = Players:GetPlayers()
    playerListContainer.CanvasSize = UDim2.new(0, 0, 0, #activePlayers * 32)
    
    for _, plr in ipairs(activePlayers) do
        if plr ~= player then
            local btn = Instance.new("TextButton", playerListContainer)
            btn.Size = UDim2.new(1, 0, 0, 28)
            btn.Position = UDim2.fromOffset(0, yOffset)
            btn.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
            btn.Text = "👤 " .. (plr.DisplayName or plr.Name) .. " (@" .. plr.Name .. ")"
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.TextSize = 10
            btn.Font = Enum.Font.Code
            btn.BorderSizePixel = 0
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)
            
            btn.MouseButton1Click:Connect(function()
                pcall(function()
                    local char = player.Character
                    if not char or not char:FindFirstChild("HumanoidRootPart") or not char:FindFirstChild("Humanoid") then return end
                    local targetChar = plr.Character
                    if not targetChar or not targetChar:FindFirstChild("HumanoidRootPart") then return end
                    
                    local nearestSeat = nil
                    local shortestDist = math.huge
                    
                    for _, obj in pairs(Workspace:GetDescendants()) do
                        if obj:IsA("VehicleSeat") or (obj:IsA("Seat") and obj.Parent:FindFirstChild("Body")) then
                            local dist = (char.HumanoidRootPart.Position - obj.Position).Magnitude
                            if dist < shortestDist then
                                shortestDist = dist
                                nearestSeat = obj
                            end
                        end
                    end
                    
                    if nearestSeat then
                        nearestSeat:Sit(char.Humanoid)
                        task.wait(0.15)
                        local carModel = nearestSeat.Parent
                        if carModel and carModel:IsA("Model") and carModel.PrimaryPart then
                            carModel:SetPrimaryPartCFrame(targetChar.HumanoidRootPart.CFrame * CFrame.new(0, 3, 0))
                        else
                            nearestSeat.CFrame = targetChar.HumanoidRootPart.CFrame * CFrame.new(0, 3, 0)
                        end
                    else
                        char.HumanoidRootPart.CFrame = targetChar.HumanoidRootPart.CFrame * CFrame.new(0, 3, 0)
                    end
                end)
            end)
            yOffset = yOffset + 32
        end
    end
end

-- RENDER LOOP (XỬ LÝ ESP, HITBOX, LASER,...)

RunService.RenderStepped:Connect(function()
    local currentLaserColor = nil
    local transparencyVal = 0.4
    if bloodLaserEnabled then 
        currentLaserColor = Color3.fromRGB(150, 0, 0)
    elseif greenLaserEnabled then 
        currentLaserColor = Color3.fromRGB(0, 255, 0)
        transparencyVal = 0
    elseif rainbowLaserEnabled then 
        currentLaserColor = Color3.fromHSV(tick() % 5 / 5, 1, 1)
        transparencyVal = 0
    end

    if player.Character then
        pcall(function()
            local tool = player.Character:FindFirstChildOfClass("Tool")
            if tool then
                local handle = tool:FindFirstChild("Handle")
                for _, obj in pairs(tool:GetDescendants()) do
                    if obj:IsA("BasePart") then
                        if currentLaserColor then
                            obj.Color = currentLaserColor
                            obj.Material = Enum.Material.Neon
                            obj.Transparency = transparencyVal
                        else
                            obj.Transparency = 0
                        end
                    elseif obj:IsA("Beam") or obj:IsA("Trail") then
                        if currentLaserColor then
                            obj.Color = ColorSequence.new(currentLaserColor)
                        end
                    end
                end

                if heavySmokeEnabled and handle then
                    if not handle:FindFirstChild("TienHeavySmoke") then
                        local smoke = Instance.new("Smoke", handle)
                        smoke.Name = "TienHeavySmoke"
                        smoke.Color = Color3.fromRGB(20, 20, 20)
                        smoke.Size = 25
                        smoke.RiseVelocity = 12
                        smoke.Opacity = 0.9
                    end
                else
                    if handle and handle:FindFirstChild("TienHeavySmoke") then
                        handle.TienHeavySmoke:Destroy()
                    end
                end
            end
        end)
    end

    if godmodEnabled and player.Character and player.Character:FindFirstChild("Humanoid") then
        player.Character.Humanoid.Health = player.Character.Humanoid.MaxHealth
    end

    if noclipEnabled and player.Character then
        for _, part in pairs(player.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end

    local humanCount = 0
    local robotCount = 0

    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character then
            local char = plr.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local head = char:FindFirstChild("Head")
            local humanoid = char:FindFirstChild("Humanoid")

            if hrp and humanoid and humanoid.Health > 0 then
                if Players:GetPlayerFromCharacter(char) then
                    humanCount = humanCount + 1
                else
                    robotCount = robotCount + 1
                end

                if head and head:IsA("BasePart") then
                    if headHitboxEnabled then
                        pcall(function()
                            head.Size = Vector3.new(headHitboxSize, headHitboxSize, headHitboxSize)
                            head.Transparency = 0.6
                            head.Material = Enum.Material.Neon
                            head.Color = Color3.fromRGB(240, 240, 240)
                            head.CanCollide = false
                        end)
                    else
                        pcall(function()
                            head.Size = Vector3.new(2, 1, 1)
                            head.Transparency = 0
                            head.Material = Enum.Material.Plastic
                            head.Color = Color3.fromRGB(248, 217, 192)
                        end)
                    end
                end

                if hitboxEnabled then
                    pcall(function()
                        hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                        hrp.Shape = Enum.PartType.Ball
                        hrp.Color = Color3.fromRGB(240, 240, 240)
                        hrp.Transparency = 0.7
                        hrp.Material = Enum.Material.Neon
                        hrp.CanCollide = false
                    end)
                else
                    pcall(function()
                        hrp.Size = Vector3.new(2, 2, 1)
                        hrp.Shape = Enum.PartType.Block
                        hrp.Transparency = 1
                    end)
                end

                if not espCache[plr] then
                    espCache[plr] = {
                        Box = Drawing.new("Square"),
                        Tracer = Drawing.new("Line"),
                        HealthBarBg = Drawing.new("Square"),
                        HealthBarFill = Drawing.new("Square")
                    }
                    espCache[plr].Box.Visible = false; espCache[plr].Box.Thickness = 1; espCache[plr].Box.Color = Color3.fromRGB(255, 255, 255); espCache[plr].Box.Filled = false
                    espCache[plr].Tracer.Visible = false; espCache[plr].Tracer.Thickness = 1; espCache[plr].Tracer.Color = Color3.fromRGB(255, 255, 255)
                    espCache[plr].HealthBarBg.Visible = false; espCache[plr].HealthBarBg.Filled = true; espCache[plr].HealthBarBg.Color = Color3.fromRGB(30, 30, 30)
                    espCache[plr].HealthBarFill.Visible = false; espCache[plr].HealthBarFill.Filled = true; espCache[plr].HealthBarFill.Color = Color3.fromRGB(255, 255, 255)
                end

                local cache = espCache[plr]
                
                local healthPercent = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
                if healthPercent < 0.3 then
                    cache.HealthBarFill.Color = Color3.fromRGB(255, 60, 60)
                else
                    cache.HealthBarFill.Color = Color3.fromRGB(255, 255, 255)
                end

                local vector, onScreen = Camera:WorldToViewportPoint(hrp.Position)

                if onScreen and anyEspActive then
                    local boxSize = Vector2.new(2000 / vector.Z, 3000 / vector.Z)
                    local boxPos = Vector2.new(vector.X - boxSize.X / 2, vector.Y - boxSize.Y / 2)

                    if espBoxesEnabled then
                        cache.Box.Visible = true; cache.Box.Size = boxSize; cache.Box.Position = boxPos
                    else
                        cache.Box.Visible = false
                    end

                    if espTracersEnabled then
                        cache.Tracer.Visible = true
                        cache.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                        cache.Tracer.To = Vector2.new(vector.X, vector.Y)
                    else
                        cache.Tracer.Visible = false
                    end

                    if espHealthEnabled then
                        local barHeight = boxSize.Y
                        local currentBarHeight = barHeight * healthPercent

                        cache.HealthBarBg.Visible = true
                        cache.HealthBarBg.Size = Vector2.new(3, barHeight)
                        cache.HealthBarBg.Position = Vector2.new(boxPos.X - 5, boxPos.Y)

                        cache.HealthBarFill.Visible = true
                        cache.HealthBarFill.Size = Vector2.new(3, currentBarHeight)
                        cache.HealthBarFill.Position = Vector2.new(boxPos.X - 5, boxPos.Y + (barHeight - currentBarHeight))
                    else
                        cache.HealthBarBg.Visible = false
                        cache.HealthBarFill.Visible = false
                    end
                else
                    cache.Box.Visible = false
                    cache.Tracer.Visible = false
                    cache.HealthBarBg.Visible = false
                    cache.HealthBarFill.Visible = false
                end
            else
                if espCache[plr] then
                    espCache[plr].Box.Visible = false
                    espCache[plr].Tracer.Visible = false
                    espCache[plr].HealthBarBg.Visible = false
                    espCache[plr].HealthBarFill.Visible = false
                end
            end
        else
            if espCache[plr] then
                espCache[plr].Box.Visible = false
                espCache[plr].Tracer.Visible = false
                espCache[plr].HealthBarBg.Visible = false
                espCache[plr].HealthBarFill.Visible = false
            end
        end
    end

    counterLabel.Text = string.format("♛ %d   ♞ %d", humanCount, robotCount)
end)
